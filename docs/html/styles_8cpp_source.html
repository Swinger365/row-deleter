32std::optional<Rgb> resolveColor(const pugi::xml_node& node, const ColorContext& ctx)
33{
34 if (!node) return std::nullopt;
35 if (xml::attribute(node, "auto").as_bool(false)) return std::nullopt;
36
37 std::optional<Rgb> base;
38 if (auto rgb = xml::attribute(node, "rgb")) {
39 base = parseHexColor(rgb.value());
40 }
41 else if (auto theme = xml::attribute(node, "theme")) {
42 const unsigned idx = theme.as_uint(~0u);
43 if (idx < ctx.theme.size()) base = ctx.theme[idx];
44 }
45 else if (auto indexed = xml::attribute(node, "indexed")) {
46 const unsigned idx = indexed.as_uint(~0u);
47 if (idx < ctx.indexed.size())
48 base = ctx.indexed[idx];
49 else
50 base = legacyIndexedColor(idx);
51 }
52 if (!base) return std::nullopt;
53
54 const double tint = xml::attribute(node, "tint").as_double(0.0);
55 return applyTint(*base, tint);
56}

80bool isRedFill(const pugi::xml_node& fill, const ColorContext& ctx, const RedCriteria& criteria)
81{
82 if (const auto pattern = xml::child(fill, "patternFill")) {
83 const std::string_view type = xml::attribute(pattern, "patternType").as_string("none");
84 if (type == "none") return false;
85 if (isRedColor(xml::child(pattern, "fgColor"), ctx, criteria)) return true;
86 return type != "solid" && isRedColor(xml::child(pattern, "bgColor"), ctx, criteria);
87 }
88 if (const auto gradient = xml::child(fill, "gradientFill")) {
89 bool red = false;
90 xml::forEachChild(gradient, "stop", [&](const pugi::xml_node& stop) {
91 red = red || isRedColor(xml::child(stop, "color"), ctx, criteria);
92 });
93 return red;
94 }
95 return false;
96}

136ThemePalette parseThemePalette(const std::string& themeXml)
137{
138 ThemePalette palette = defaultThemePalette();
139 if (themeXml.empty()) return palette;
140
141 pugi::xml_document doc;
142 xml::load(doc, themeXml, "тема");
143 const auto scheme = xml::child(xml::child(doc.document_element(), "themeElements"), "clrScheme");
144 if (!scheme) return palette;
145
146 static const std::pair<std::string_view, std::size_t> kSlots[] = {
147 {"lt1", 0}, {"dk1", 1}, {"lt2", 2}, {"dk2", 3}, {"accent1", 4}, {"accent2", 5},
148 {"accent3", 6}, {"accent4", 7}, {"accent5", 8}, {"accent6", 9}, {"hlink", 10}, {"folHlink", 11},
149 };
150 for (const auto& [name, index] : kSlots) {
151 const auto slot = xml::child(scheme, name);
152 std::optional<Rgb> color;
153 if (const auto srgb = xml::child(slot, "srgbClr")) {
154 color = parseHexColor(xml::attribute(srgb, "val").value());
155 }
156 else if (const auto sys = xml::child(slot, "sysClr")) {
157 color = parseHexColor(xml::attribute(sys, "lastClr").value());
158 if (!color) {
159 const std::string_view val = xml::attribute(sys, "val").value();
160 if (val == "windowText") color = Rgb{0, 0, 0};
161 if (val == "window") color = Rgb{0xFF, 0xFF, 0xFF};
162 }
163 }
164 if (color) palette[index] = *color;
165 }
166 return palette;
167}

171RedStyleTable RedStyleTable::build(const std::string& stylesXml, const std::string& themeXml, const StyleOptions& options)
172{
173 if (stylesXml.empty()) return RedStyleTable{};
174
175 pugi::xml_document doc;
176 xml::load(doc, stylesXml, "styles.xml");
177 const auto root = doc.document_element();
178
179 ColorContext ctx;
180 ctx.theme = parseThemePalette(themeXml);
181 xml::forEachChild(xml::child(xml::child(root, "colors"), "indexedColors"), "rgbColor", [&](const pugi::xml_node& c) {
182 ctx.indexed.push_back(parseHexColor(xml::attribute(c, "rgb").value()).value_or(Rgb{}));
183 });
184
185 const RedCriteria& criteria = options.criteria;
186 const bool checkFill = options.target != ColorTarget::Font;
187 const bool checkFont = options.target != ColorTarget::Fill;
188
189 std::vector<bool> redFills, redFonts;
190 if (checkFill)
191 redFills = evaluateList(xml::child(root, "fills"), "fill",
192 [&](const pugi::xml_node& fill) { return isRedFill(fill, ctx, criteria); });
193 if (checkFont)
194 redFonts = evaluateList(xml::child(root, "fonts"), "font", [&](const pugi::xml_node& font) {
195 return isRedColor(xml::child(font, "color"), ctx, criteria);
196 });
197
198 std::vector<bool> red = evaluateList(xml::child(root, "cellXfs"), "xf", [&](const pugi::xml_node& xf) {
199 return (checkFill && flagAt(redFills, xml::attribute(xf, "fillId"))) ||
200 (checkFont && flagAt(redFonts, xml::attribute(xf, "fontId")));
201 });
202 return RedStyleTable(std::move(red));
203}
