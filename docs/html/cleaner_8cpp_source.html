56std::vector<SheetTask> resolveSheets(const XlsxPackage& package, const CleanOptions& options)
57{
58 std::vector<SheetTask> tasks;
59 std::set<std::string> seen;
60 auto add = [&](const SheetInfo& info, std::uint32_t startRow) {
61 if (!seen.insert(info.name).second) throw std::invalid_argument("Лист \"" + info.name + "\" указан дважды");
62 tasks.push_back({info, std::make_unique<SheetProcessor>(info.name, startRow), {}});
63 };
64
65 if (options.allSheets) {
66 for (const auto& info : package.worksheets()) add(info, options.defaultStartRow);
67 return tasks;
68 }
69 if (options.sheets.empty()) throw std::invalid_argument("Не указан ни один лист");
70
71 for (const auto& spec : options.sheets) {
72 const auto info = package.findWorksheet(spec.name);
73 if (!info) {
74 std::string available;
75 for (const auto& ws : package.worksheets()) available += (available.empty() ? "" : ", ") + ("\"" + ws.name + "\"");
76 throw std::invalid_argument("Лист \"" + spec.name + "\" не найден. Доступные листы: " + available);
77 }
78 add(*info, spec.startRow.value_or(options.defaultStartRow));
79 }
80 return tasks;
81}

85CleanReport cleanWorkbook(const CleanOptions& options)
86{
87 if (options.inputPath.empty()) throw std::invalid_argument("Не указан входной файл");
88 if (!options.dryRun && options.outputPath.empty()) throw std::invalid_argument("Не указан выходной файл");
89 if (options.defaultStartRow == 0) throw std::invalid_argument("Номер начальной строки должен быть >= 1");
90
91 CleanReport report;
92 StageTimer timer(report);
93
94 XlsxPackage package(options.inputPath);
95 std::vector<SheetTask> tasks = resolveSheets(package, options);
96 timer.mark("открытие архива");
97
98 const RedStyleTable styles =
99 RedStyleTable::build(package.read(package.stylesPath()), package.read(package.themePath()), options.style);
100 report.styleCount = styles.styleCount();
101 report.redStyleCount = styles.redCount();
102
103 ThreadPool pool(options.threads);
104 report.threadsUsed = pool.size();
105
106 {
107 std::vector<std::future<void>> futures;
108 for (auto& task : tasks)
109 futures.push_back(pool.submit([&task, &package] { task.processor->load(package.read(task.info.path)); }));
110 waitAll(futures);
111 }
112 timer.mark("распаковка и разбор XML");
113
114 if (report.redStyleCount > 0) {
115 std::vector<std::future<void>> futures;
116 const std::size_t grain = std::max<std::size_t>(1, options.rowsPerTask);
117 for (auto& task : tasks) {
118 SheetProcessor* processor = task.processor.get();
119 for (std::size_t begin = 0; begin < processor->candidateRowCount(); begin += grain)
120 futures.push_back(
121 pool.submit([processor, begin, grain, &styles] { processor->detect(begin, begin + grain, styles); }));
122 }
123 waitAll(futures);
124 }
125 timer.mark("поиск красных строк");
126
127 {
128 std::vector<std::future<void>> futures;
129 const bool dryRun = options.dryRun;
130 for (auto& task : tasks)
131 futures.push_back(pool.submit([&task, dryRun] {
132 task.processor->apply();
133 if (!dryRun && task.processor->modified()) task.resultXml = task.processor->serialize();
134 }));
135 waitAll(futures);
136 }
137 timer.mark("очистка и сериализация");
138
139 bool formulasRemoved = false;
140 bool anyModified = false;
141 for (auto& task : tasks) {
142 report.sheets.push_back({task.processor->name(), task.processor->startRow(), task.processor->stats()});
143 formulasRemoved = formulasRemoved || task.processor->stats().formulasRemoved > 0;
144 if (task.processor->modified()) {
145 anyModified = true;
146 if (!options.dryRun) package.write(task.info.path, std::move(task.resultXml));
147 }
148 }
149
150 if (options.dryRun) return report;
151
152 if (formulasRemoved) report.calcChainRemoved = package.removeCalcChain();
153 if (anyModified || options.outputPath != options.inputPath) {
154 package.saveAs(options.outputPath, pool, options.compressionLevel);
155 report.written = true;
156 }
157 timer.mark("сжатие и запись");
158 return report;
159}
