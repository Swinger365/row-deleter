47bool clearCell(pugi::xml_node cell, bool& touched)
48{
49 bool hadContent = false;
50 for (pugi::xml_node node = cell.first_child(); node;) {
51 const pugi::xml_node next = node.next_sibling();
52 if (node.type() == pugi::node_element) hadContent = true;
53 cell.remove_child(node);
54 touched = true;
55 node = next;
56 }
57 for (pugi::xml_attribute attr = cell.first_attribute(); attr;) {
58 const pugi::xml_attribute next = attr.next_attribute();
59 const std::string_view name = attr.name();
60 if (name != "r" && name != "s") {
61 cell.remove_attribute(attr);
62 touched = true;
63 }
64 attr = next;
65 }
66 return hadContent;
67}

101void SheetProcessor::detect(std::size_t begin, std::size_t end, const RedStyleTable& styles)
102{
103 for (std::size_t i = begin; i < end && i < m_rows.size(); ++i) {
104 for (pugi::xml_node cell = xml::child(m_rows[i].node, "c"); cell; cell = xml::nextSibling(cell, "c")) {
105 if (styles.isRed(xml::attribute(cell, "s").as_uint(0))) {

113void SheetProcessor::apply()
114{
115 std::unordered_map<std::string, SharedMaster> clearedMasters;
116
117 for (std::size_t i = 0; i < m_rows.size(); ++i) {
118 if (!m_redFlags[i]) continue;
119 const RowRef& row = m_rows[i];
120 m_stats.redRows.push_back(row.number);
121
122 forEachCell(row.node, [&](pugi::xml_node cell, std::uint32_t col) {
123 const pugi::xml_node f = xml::child(cell, "f");
124 if (f) {
125 ++m_stats.formulasRemoved;
126 if (isSharedFormula(f) && xml::attribute(f, "ref") && *f.child_value())
127 clearedMasters[xml::attribute(f, "si").value()] = {f.child_value(), {row.number, col}};
128 }
129 if (clearCell(cell, m_modified)) ++m_stats.cellsCleared;
130 });
131 }
132
133 if (clearedMasters.empty()) return;
134
135 std::uint32_t number = 0;
136 xml::forEachChild(m_sheetData, "row", [&](const pugi::xml_node& row) {
137 const unsigned r = xml::attribute(row, "r").as_uint(0);
138 number = r != 0 ? r : number + 1;
139 forEachCell(row, [&](pugi::xml_node cell, std::uint32_t col) {
140 pugi::xml_node f = xml::child(cell, "f");
141 if (!isSharedFormula(f)) return;
142 const auto it = clearedMasters.find(xml::attribute(f, "si").value());
143 if (it == clearedMasters.end()) return;
144
145 const SharedMaster& master = it->second;
146 const std::string formula =
147 shiftFormula(master.formula, static_cast<std::int64_t>(number) - master.pos.row,
148 static_cast<std::int64_t>(col) - master.pos.col);
149 for (const char* attrName : {"t", "si", "ref"}) f.remove_attribute(attrName);
150 f.text().set(formula.c_str());
151 ++m_stats.sharedFormulasExpanded;
152 m_modified = true;
153 });
154 });
155}
