70 template <typename Fn>
71 auto submit(Fn&& fn) -> std::future<std::invoke_result_t<std::decay_t<Fn>>>
72 {
73 using Result = std::invoke_result_t<std::decay_t<Fn>>;
74 auto task = std::make_shared<std::packaged_task<Result()>>(std::forward<Fn>(fn));
75 std::future<Result> future = task->get_future();
76 {
77 std::lock_guard lock(m_mutex);
78 if (m_stopping) throw std::runtime_error("ThreadPool: submit после остановки");
79 m_tasks.emplace([task] { (*task)(); });
80 }
81 m_cv.notify_one();
82 return future;
83 }

85private:
87 void workerLoop()
88 {
89 for (;;) {
90 std::function<void()> task;
91 {
92 std::unique_lock lock(m_mutex);
93 m_cv.wait(lock, [this] { return m_stopping || !m_tasks.empty(); });
94 if (m_stopping && m_tasks.empty()) return;
95 task = std::move(m_tasks.front());
96 m_tasks.pop();
97 }
98 task();
99 }
100 }

133void parallelFor(ThreadPool& pool, std::size_t count, std::size_t grain, Fn&& fn)
134{
135 if (count == 0) return;
136 grain = std::max<std::size_t>(1, grain);

138 futures.reserve((count + grain - 1) / grain);
139 for (std::size_t begin = 0; begin < count; begin += grain) {
140 const std::size_t end = std::min(count, begin + grain);

142 }
