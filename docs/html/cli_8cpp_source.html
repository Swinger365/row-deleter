28T parseNumber(const std::string& text, const std::string& option, T minValue, T maxValue)
29{
30 T value{};
31 const char* begin = text.data();
32 const char* end = text.data() + text.size();
33 const auto [ptr, ec] = std::from_chars(begin, end, value);
34 if (ec != std::errc{} || ptr != end || value < minValue || value > maxValue)
35 throw CliError("Некорректное значение для " + option + ": \"" + text + "\"");
36 return value;
37}

48double parseDouble(const std::string& text, const std::string& option, double minValue, double maxValue)
49{
50 try {
51 std::size_t used = 0;
52 const double value = std::stod(text, &used);
53 if (used == text.size() && value >= minValue && value <= maxValue) return value;
54 }
55 catch (const std::exception&) {
56 }
57 throw CliError("Некорректное значение для " + option + ": \"" + text + "\"");
58}

68SheetSpec parseSheetSpec(const std::string& text)
69{
70 SheetSpec spec;
71 const auto colon = text.rfind(':');
72 if (colon == std::string::npos) {
73 spec.name = text;
74 }
75 else {
76 spec.name = text.substr(0, colon);
77 spec.startRow = parseNumber<std::uint32_t>(text.substr(colon + 1), "--sheet", 1, 1048576);
78 }
79 if (spec.name.empty()) throw CliError("Пустое имя листа в --sheet \"" + text + "\"");
80 return spec;
81}

89std::vector<Rgb> parseColorList(const std::string& text)
90{
91 std::vector<Rgb> colors;
92 std::size_t pos = 0;
93 while (pos <= text.size()) {
94 const auto comma = text.find(',', pos);
95 const std::string item = text.substr(pos, comma == std::string::npos ? std::string::npos : comma - pos);
96 const auto color = parseHexColor(item);
97 if (!color) throw CliError("Некорректный цвет \"" + item + "\" (ожидается RRGGBB)");
98 colors.push_back(*color);
99 if (comma == std::string::npos) break;
100 pos = comma + 1;
101 }
102 return colors;
103}

121CliCommand parseCommandLine(const std::vector<std::string>& args)
122{
123 CliCommand cmd;
124 CleanOptions& opt = cmd.options;
125 bool inPlace = false;
126
127 for (std::size_t i = 0; i < args.size(); ++i) {
128 std::string arg = args[i];
129 std::optional<std::string> inlineValue;
130 if (arg.rfind("--", 0) == 0) {
131 const auto eq = arg.find('=');
132 if (eq != std::string::npos) {
133 inlineValue = arg.substr(eq + 1);
134 arg = arg.substr(0, eq);
135 }
136 }
137 auto value = [&]() -> std::string {
138 if (inlineValue) return *inlineValue;
139 if (i + 1 >= args.size()) throw CliError("Для " + arg + " нужно значение");
140 return args[++i];
141 };
142 auto noValue = [&] {
143 if (inlineValue) throw CliError("Опция " + arg + " не принимает значение");
144 };
145
146 if (arg == "-h" || arg == "--help") {
147 noValue();
148 cmd.showHelp = true;
149 }
150 else if (arg == "-i" || arg == "--input") opt.inputPath = value();
151 else if (arg == "-o" || arg == "--output") opt.outputPath = value();
152 else if (arg == "--in-place") { noValue(); inPlace = true; }
153 else if (arg == "-s" || arg == "--sheet") opt.sheets.push_back(parseSheetSpec(value()));
154 else if (arg == "--all-sheets") { noValue(); opt.allSheets = true; }
155 else if (arg == "-r" || arg == "--start-row")
156 opt.defaultStartRow = parseNumber<std::uint32_t>(value(), arg, 1, 1048576);
157 else if (arg == "-j" || arg == "--threads")
158 opt.threads = parseNumber<std::size_t>(value(), arg, 1, 1024);
159 else if (arg == "--rows-per-task")
160 opt.rowsPerTask = parseNumber<std::size_t>(value(), arg, 1, std::numeric_limits<std::uint32_t>::max());
161 else if (arg == "--compression-level")
162 opt.compressionLevel = parseNumber<int>(value(), arg, 1, 9);
163 else if (arg == "-t" || arg == "--target") opt.style.target = parseTarget(value());
164 else if (arg == "--colors") opt.style.criteria.exactColors = parseColorList(value());
165 else if (arg == "--hue-tolerance") opt.style.criteria.maxHueDeviation = parseDouble(value(), arg, 0.0, 180.0);
166 else if (arg == "--min-saturation") opt.style.criteria.minSaturation = parseDouble(value(), arg, 0.0, 1.0);
167 else if (arg == "--min-value") opt.style.criteria.minValue = parseDouble(value(), arg, 0.0, 1.0);
168 else if (arg == "--dry-run") { noValue(); opt.dryRun = true; }
169 else if (arg == "-v" || arg == "--verbose") { noValue(); cmd.verbose = true; }
170 else if (!arg.empty() && arg.front() == '-') throw CliError("Неизвестная опция: " + arg);
171 else if (opt.inputPath.empty()) opt.inputPath = arg;
172 else throw CliError("Лишний аргумент: " + arg);
173 }
174
175 if (cmd.showHelp) return cmd;
176
177 if (opt.inputPath.empty()) throw CliError("Не указан входной файл (-i)");
178 if (opt.allSheets && !opt.sheets.empty()) throw CliError("--all-sheets нельзя совмещать с --sheet");
179 if (!opt.allSheets && opt.sheets.empty()) throw CliError("Укажите листы через --sheet (или --all-sheets)");
180 if (inPlace && !opt.outputPath.empty()) throw CliError("--in-place нельзя совмещать с --output");
181 if (inPlace) opt.outputPath = opt.inputPath;
182 if (opt.outputPath.empty() && !opt.dryRun) throw CliError("Укажите выходной файл (-o), --in-place или --dry-run");
183 return cmd;
184}

187{
188 return "Использование: " + programName + R"( -i ВХОД.xlsx (-o ВЫХОД.xlsx | --in-place | --dry-run) -s ЛИСТ[:СТРОКА]... [опции]
189
190Очищает содержимое (значения и формулы, форматирование сохраняется) всех строк, начиная
191с заданной, в которых есть хотя бы одна ячейка красного цвета.
192
193Файлы и листы:
194 -i, --input FILE входной файл .xlsx (можно указать позиционно)
195 -o, --output FILE куда сохранить результат
196 --in-place перезаписать входной файл
197 -s, --sheet NAME[:ROW] лист для обработки, можно повторять; ROW — своя начальная строка
198 --all-sheets обработать все рабочие листы
199 -r, --start-row N начальная строка по умолчанию (1)
200
201Что считать красным:
202 -t, --target WHAT fill (заливка, по умолчанию) | font (цвет шрифта) | any
203 --colors LIST точный список цветов RRGGBB через запятую, например FF0000,C00000
204 --hue-tolerance D допуск оттенка от чистого красного в градусах (20)
205 --min-saturation X минимальная насыщенность 0..1 (0.5)
206 --min-value X минимальная яркость 0..1 (0.5)
207
208Производительность и прочее:
209 -j, --threads N число потоков (по умолчанию — число ядер)
210 --rows-per-task N строк в одной задаче поиска (4096)
211 --compression-level N уровень сжатия 1..9 (6); 1 — заметно быстрее, файл чуть больше
212 --dry-run только показать, какие строки будут очищены
213 -v, --verbose вывести номера найденных строк и время этапов
214 -h, --help эта справка
215)";
