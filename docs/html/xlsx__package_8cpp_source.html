59std::string resolveTarget(const std::string& baseDir, std::string_view target)
60{
61 std::string joined = (!target.empty() && target.front() == '/') ? std::string(target.substr(1))
62 : baseDir + std::string(target);
63 std::vector<std::string> parts;
64 std::size_t pos = 0;
65 while (pos <= joined.size()) {
66 const auto next = joined.find('/', pos);
67 const std::string part = joined.substr(pos, next == std::string::npos ? std::string::npos : next - pos);
68 if (part == "..") {
69 if (!parts.empty()) parts.pop_back();
70 }
71 else if (!part.empty() && part != ".") {
72 parts.push_back(part);
73 }
74 if (next == std::string::npos) break;
75 pos = next + 1;
76 }
77 std::string out;
78 for (const auto& p : parts) out += (out.empty() ? "" : "/") + p;
79 return out;
80}

97std::vector<Relationship> parseRelationships(const std::string& data, const std::string& what)
98{

104 if (std::string_view(xml::attribute(rel, "TargetMode").value()) == "External") return;
105 rels.push_back({xml::attribute(rel, "Id").value(), xml::attribute(rel, "Type").value(),

132XlsxPackage::XlsxPackage(const std::string& path) : m_zip(openZip(path))
133{
134 for (const auto& rel : parseRelationships(read("_rels/.rels"), "_rels/.rels"))
135 if (endsWith(rel.type, "/officeDocument")) m_workbookPath = resolveTarget("", rel.target);
136 if (m_workbookPath.empty() || !m_zip.contains(m_workbookPath))
137 throw std::runtime_error("\"" + path + "\" не похож на книгу Excel: не найден workbook.xml");
138
139 m_workbookRelsPath = relsPathFor(m_workbookPath);
140 const std::string baseDir = directoryOf(m_workbookPath);
141 std::unordered_map<std::string, Relationship> relById;
142 for (auto& rel : parseRelationships(read(m_workbookRelsPath), m_workbookRelsPath)) {
143 rel.target = resolveTarget(baseDir, rel.target);
144 if (endsWith(rel.type, "/styles")) m_stylesPath = rel.target;
145 if (endsWith(rel.type, "/theme")) m_themePath = rel.target;
146 if (endsWith(rel.type, "/calcChain")) m_calcChainPath = rel.target;
147 relById.emplace(rel.id, rel);
148 }
149
150 pugi::xml_document workbook;
151 xml::load(workbook, read(m_workbookPath), m_workbookPath);
152 xml::forEachChild(xml::child(workbook.document_element(), "sheets"), "sheet", [&](const pugi::xml_node& sheet) {
153 const std::string name = xml::attribute(sheet, "name").value();
154 const auto it = relById.find(xml::attribute(sheet, "id").value());
155 if (it != relById.end() && endsWith(it->second.type, "/worksheet"))
156 m_worksheets.push_back({name, it->second.target});
157 else
158 m_otherSheetNames.push_back(name);
159 });
160}

185bool XlsxPackage::removeCalcChain()
186{
187 if (m_calcChainPath.empty() || !m_zip.contains(m_calcChainPath)) return false;
188 m_removed.insert(m_calcChainPath);
189 m_replaced.erase(m_calcChainPath);
190
191 pugi::xml_document rels;
192 xml::load(rels, read(m_workbookRelsPath), m_workbookRelsPath);
193 pugi::xml_node relsRoot = rels.document_element();
194 for (pugi::xml_node rel = xml::child(relsRoot, "Relationship"); rel;) {
195 const pugi::xml_node next = xml::nextSibling(rel, "Relationship");
196 if (endsWith(xml::attribute(rel, "Type").value(), "/calcChain")) relsRoot.remove_child(rel);
197 rel = next;
198 }
199 write(m_workbookRelsPath, xml::serialize(rels));
200
201 const std::string contentTypesPath = "[Content_Types].xml";
202 pugi::xml_document types;
203 xml::load(types, read(contentTypesPath), contentTypesPath);
204 pugi::xml_node typesRoot = types.document_element();
205 const std::string partName = "/" + m_calcChainPath;
206 for (pugi::xml_node node = xml::child(typesRoot, "Override"); node;) {
207 const pugi::xml_node next = xml::nextSibling(node, "Override");
208 if (partName == xml::attribute(node, "PartName").value()) typesRoot.remove_child(node);
209 node = next;
210 }
211 write(contentTypesPath, xml::serialize(types));
212
213 m_calcChainPath.clear();
214 return true;
215}

217void XlsxPackage::saveAs(const std::string& path, ThreadPool& pool, int compressionLevel) const
218{
219 struct Chunk
220 {
221 const std::string* data;
222 std::size_t begin;
223 std::size_t end;
224 bool last;
225 };
226 std::vector<const std::string*> parts;
227 std::vector<std::vector<std::size_t>> chunksOfPart;
228 std::vector<Chunk> chunks;
229 for (const auto& [name, data] : m_replaced) {
230 chunksOfPart.emplace_back();
231 parts.push_back(&data);
232 for (std::size_t begin = 0;;) {
233 const std::size_t end = std::min(data.size(), begin + kChunkSize);
234 chunksOfPart.back().push_back(chunks.size());
235 chunks.push_back({&data, begin, end, end == data.size()});
236 if (end == data.size()) break;
237 begin = end;
238 }
239 }
240
241 std::vector<std::string> deflated(chunks.size());
242 std::vector<std::uint32_t> crcs(parts.size());
243 parallelFor(pool, chunks.size() + parts.size(), 1, [&](std::size_t begin, std::size_t end) {
244 for (std::size_t i = begin; i < end; ++i) {
245 if (i < parts.size()) {
246 crcs[i] = crc32(parts[i]->data(), parts[i]->size());
247 }
248 else {
249 const Chunk& c = chunks[i - parts.size()];
250 deflated[i - parts.size()] = deflateChunk(c.data->data() + c.begin, c.end - c.begin, compressionLevel, c.last);
251 }
252 }
253 });
254
255 std::map<std::string, CompressedEntry> compressed;
256 std::size_t index = 0;
257 for (const auto& [name, data] : m_replaced) {
258 CompressedEntry entry;
259 entry.uncompressedSize = data.size();
260 entry.crc32 = crcs[index];
261 for (const std::size_t c : chunksOfPart[index]) {
262 entry.deflated += deflated[c];
263 std::string().swap(deflated[c]);
264 }
265 compressed.emplace(name, std::move(entry));
266 ++index;
267 }
268
269 ZipWriter writer;
270 std::set<std::string> written;
271 for (const auto& name : m_zip.entryNames()) {
272 if (m_removed.count(name) || !written.insert(name).second) continue;
273 if (const auto it = compressed.find(name); it != compressed.end())
274 writer.addCompressed(name, it->second);
275 else
276 writer.copyFrom(m_zip, name);
277 }
278 for (const auto& [name, entry] : compressed)
279 if (!written.count(name)) writer.addCompressed(name, entry);
280 writer.finish(path);
281}
