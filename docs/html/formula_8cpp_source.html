42std::optional<RefPart> parseColumnPart(std::string_view s)
43{
44 RefPart part;
45 if (!s.empty() && s.front() == '$') {
46 part.absolute = true;
47 s.remove_prefix(1);
48 }
49 if (s.empty() || s.size() > 3) return std::nullopt;
50 for (char c : s)
51 if (!isLetter(c)) return std::nullopt;
52 part.value = columnFromLetters(s);
53 if (part.value == 0) return std::nullopt;
54 return part;

62std::optional<RefPart> parseRowPart(std::string_view s)
63{
64 RefPart part;
65 if (!s.empty() && s.front() == '$') {
66 part.absolute = true;
67 s.remove_prefix(1);
68 }
69 if (s.empty() || s.size() > 7) return std::nullopt;
70 std::uint64_t value = 0;
71 for (char c : s) {
72 if (!isDigit(c)) return std::nullopt;

75 if (value == 0 || value > kMaxRows) return std::nullopt;

92std::optional<CellRefParts> parseCellParts(std::string_view s)
93{
94 std::size_t i = 0;
95 if (i < s.size() && s[i] == '$') ++i;
96 while (i < s.size() && isLetter(s[i])) ++i;
97 const auto col = parseColumnPart(s.substr(0, i));
98 if (!col) return std::nullopt;
99 const auto row = parseRowPart(s.substr(i));
100 if (!row) return std::nullopt;
101 return CellRefParts{*col, *row};

160std::size_t skipBrackets(std::string_view s, std::size_t i)
161{
162 int depth = 0;
163 while (i < s.size()) {
164 if (s[i] == '[') ++depth;

180std::uint32_t columnFromLetters(std::string_view letters)
181{
182 if (letters.empty() || letters.size() > 3) return 0;
183 std::uint32_t value = 0;
184 for (char c : letters) {
185 if (!isLetter(c)) return 0;
186 value = value * 26 + static_cast<std::uint32_t>(std::toupper(static_cast<unsigned char>(c)) - 'A' + 1);
187 }
188 return value <= kMaxColumns ? value : 0;
189}

191std::string columnToLetters(std::uint32_t column)
192{
193 std::string out;
194 while (column > 0) {
195 const std::uint32_t rem = (column - 1) % 26;
196 out.insert(out.begin(), static_cast<char>('A' + rem));
197 column = (column - 1) / 26;
198 }
199 return out;
200}

209std::string shiftFormula(std::string_view f, std::int64_t dRow, std::int64_t dCol)
210{
211 std::string out;
212 out.reserve(f.size() + 8);
213
214 std::size_t i = 0;
215 while (i < f.size()) {
216 const char c = f[i];
217 if (c == '"' || c == '\'') {
218 const std::size_t end = skipQuoted(f, i, c);
219 out.append(f.substr(i, end - i));
220 i = end;
221 continue;
222 }
223 if (c == '[') {
224 const std::size_t end = skipBrackets(f, i);
225 out.append(f.substr(i, end - i));
226 i = end;
227 continue;
228 }
229 if (!isWordChar(c)) {
230 out.push_back(c);
231 ++i;
232 continue;
233 }
234
235 const std::size_t end = wordEnd(f, i);
236 const std::string_view word = f.substr(i, end - i);
237 const char next = end < f.size() ? f[end] : '\0';
238
239 if (next == '(' || next == '!') {
240 out.append(word);
241 i = end;
242 continue;
243 }
244
245 if (const auto cell = parseCellParts(word)) {
246 const auto col = shifted(cell->col, dCol, kMaxColumns);
247 const auto row = shifted(cell->row, dRow, kMaxRows);
248 if (col && row)
249 out += formatColumn(cell->col, *col) + formatRow(cell->row, *row);
250 else
251 out += "#REF!";
252 i = end;
253 continue;
254 }
255
256 if (next == ':' && end + 1 < f.size() && isWordChar(f[end + 1])) {
257 const std::size_t end2 = wordEnd(f, end + 1);
258 const std::string_view word2 = f.substr(end + 1, end2 - end - 1);
259 const char after = end2 < f.size() ? f[end2] : '\0';
260 if (after != '(' && after != '!') {
261 const auto c1 = parseColumnPart(word), c2 = parseColumnPart(word2);
262 if (c1 && c2) {
263 const auto s1 = shifted(*c1, dCol, kMaxColumns), s2 = shifted(*c2, dCol, kMaxColumns);
264 out += (s1 && s2) ? formatColumn(*c1, *s1) + ":" + formatColumn(*c2, *s2) : "#REF!";
265 i = end2;
266 continue;
267 }
268 const auto r1 = parseRowPart(word), r2 = parseRowPart(word2);
269 if (r1 && r2) {
270 const auto s1 = shifted(*r1, dRow, kMaxRows), s2 = shifted(*r2, dRow, kMaxRows);
271 out += (s1 && s2) ? formatRow(*r1, *s1) + ":" + formatRow(*r2, *s2) : "#REF!";
272 i = end2;
273 continue;
274 }
275 }
276 }
277
278 out.append(word);
279 i = end;
280 }
281 return out;
282}
