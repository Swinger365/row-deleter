31std::string formatRows(const std::vector<std::uint32_t>& rows)
32{
33 std::string out;
34 for (std::size_t i = 0; i < rows.size();) {
35 std::size_t j = i;
36 while (j + 1 < rows.size() && rows[j + 1] == rows[j] + 1) ++j;
37 if (!out.empty()) out += ", ";
38 out += std::to_string(rows[i]);
39 if (j > i) out += "-" + std::to_string(rows[j]);
40 i = j + 1;
41 }
42 return out;
43}
