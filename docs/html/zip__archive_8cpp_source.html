52std::string deflateChunk(const char* data, std::size_t size, int level, bool last)
53{
54 const mz_uint flags = tdefl_create_comp_flags_from_zip_params(level, -MZ_DEFAULT_WINDOW_BITS, MZ_DEFAULT_STRATEGY);
55 std::unique_ptr<tdefl_compressor> compressor(new tdefl_compressor);
56
57 std::string out;
58 out.reserve(size / 4 + 64);
59 const tdefl_put_buf_func_ptr append = [](const void* buf, int len, void* user) -> mz_bool {
60 static_cast<std::string*>(user)->append(static_cast<const char*>(buf), static_cast<std::size_t>(len));
61 return MZ_TRUE;
62 };
63 if (tdefl_init(compressor.get(), append, &out, static_cast<int>(flags)) != TDEFL_STATUS_OKAY)
64 throw std::runtime_error("ошибка инициализации deflate");
65 const tdefl_status expected = last ? TDEFL_STATUS_DONE : TDEFL_STATUS_OKAY;
66 if (tdefl_compress_buffer(compressor.get(), data, size, last ? TDEFL_FINISH : TDEFL_FULL_FLUSH) != expected)
67 throw std::runtime_error("ошибка сжатия deflate");
68 return out;
69}

87ZipReader::ZipReader(const std::string& path) : m_impl(std::make_unique<Impl>())
88{
89 std::ifstream in(fs::u8path(path), std::ios::binary);
90 if (!in) throw std::runtime_error("не удалось открыть файл \"" + path + "\"");
91 in.seekg(0, std::ios::end);
92 m_buffer.resize(static_cast<std::size_t>(in.tellg()));
93 in.seekg(0);
94 in.read(m_buffer.data(), static_cast<std::streamsize>(m_buffer.size()));
95 if (!in) throw std::runtime_error("ошибка чтения \"" + path + "\"");
96
97 initReader(m_impl->zip, m_buffer);
98 const mz_uint count = mz_zip_reader_get_num_files(&m_impl->zip);
99 m_names.reserve(count);
100 for (mz_uint i = 0; i < count; ++i) {
101 mz_zip_archive_file_stat stat;
102 if (!mz_zip_reader_file_stat(&m_impl->zip, i, &stat)) throw std::runtime_error("повреждённый zip-архив");
103 m_names.emplace_back(stat.m_filename);
104 }
105}

117{
118 mz_zip_archive zip;

122 mz_zip_archive& z;

125
126 const int index = mz_zip_reader_locate_file(&zip, name.c_str(), nullptr, 0);

196void ZipWriter::finish(const std::string& path)
197{
198 void* buffer = nullptr;
199 std::size_t size = 0;
200 m_impl->check(mz_zip_writer_finalize_heap_archive(&m_impl->zip, &buffer, &size), "ошибка финализации архива");
201 std::unique_ptr<void, FreeDeleter> data(buffer);
202
203 const fs::path target = fs::u8path(path);
204 fs::path temp = target;
205 temp += ".tmp" + std::to_string(std::random_device{}());
206 {
207 std::ofstream out(temp, std::ios::binary | std::ios::trunc);
208 out.write(static_cast<const char*>(data.get()), static_cast<std::streamsize>(size));
209 out.close();
210 if (!out) {
211 std::error_code ec;
212 fs::remove(temp, ec);
213 throw std::runtime_error("не удалось записать \"" + path + "\"");
214 }
215 }
216 std::error_code ec;
217 fs::rename(temp, target, ec);
218 if (ec) {
219 fs::remove(target, ec);
220 fs::rename(temp, target, ec);
221 }
222 if (ec) {
223 fs::remove(temp, ec);
224 throw std::runtime_error("не удалось сохранить \"" + path + "\": " + ec.message());
225 }
226}
