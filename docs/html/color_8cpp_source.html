51Hls toHls(const Rgb& c)
52{
53 const double r = c.r / 255.0, g = c.g / 255.0, b = c.b / 255.0;
54 const double maxC = std::max({r, g, b});
55 const double minC = std::min({r, g, b});
56 Hls out;
57 out.l = (maxC + minC) / 2.0;
58 const double delta = maxC - minC;
59 if (delta == 0.0) return out;
60
61 out.s = out.l <= 0.5 ? delta / (maxC + minC) : delta / (2.0 - maxC - minC);
62 double h = 0.0;
63 if (maxC == r)
64 h = (g - b) / delta;
65 else if (maxC == g)
66 h = 2.0 + (b - r) / delta;
67 else
68 h = 4.0 + (r - g) / delta;
69 h /= 6.0;
70 if (h < 0.0) h += 1.0;
71 out.h = h;
72 return out;
73}

82double hueToChannel(double m1, double m2, double h)
83{
84 if (h < 0.0) h += 1.0;
85 if (h > 1.0) h -= 1.0;
86 if (h < 1.0 / 6.0) return m1 + (m2 - m1) * h * 6.0;
87 if (h < 0.5) return m2;
88 if (h < 2.0 / 3.0) return m1 + (m2 - m1) * (2.0 / 3.0 - h) * 6.0;
89 return m1;
90}

97Rgb fromHls(const Hls& c)
98{
99 if (c.s == 0.0) {
100 const auto v = toByte(c.l);
101 return {v, v, v};
102 }
103 const double m2 = c.l <= 0.5 ? c.l * (1.0 + c.s) : c.l + c.s - c.l * c.s;
104 const double m1 = 2.0 * c.l - m2;
105 return {toByte(hueToChannel(m1, m2, c.h + 1.0 / 3.0)), toByte(hueToChannel(m1, m2, c.h)),
106 toByte(hueToChannel(m1, m2, c.h - 1.0 / 3.0))};
107}

123std::optional<Rgb> parseHexColor(std::string_view text)
124{
125 if (!text.empty() && text.front() == '#') text.remove_prefix(1);
126 if (text.size() == 8)
127 text.remove_prefix(2);
128 else if (text.size() != 6)
129 return std::nullopt;
130
131 std::array<std::uint8_t, 3> bytes{};
132 for (std::size_t i = 0; i < 3; ++i) {
133 const int hi = hexDigit(text[2 * i]);
134 const int lo = hexDigit(text[2 * i + 1]);
135 if (hi < 0 || lo < 0) return std::nullopt;
136 bytes[i] = static_cast<std::uint8_t>(hi * 16 + lo);
137 }
138 return Rgb{bytes[0], bytes[1], bytes[2]};
139}

141std::string toHex(const Rgb& color)
142{
143 static constexpr char digits[] = "0123456789ABCDEF";
144 std::string out(6, '0');
145 const std::uint8_t channels[] = {color.r, color.g, color.b};
146 for (std::size_t i = 0; i < 3; ++i) {
147 out[2 * i] = digits[channels[i] >> 4];
148 out[2 * i + 1] = digits[channels[i] & 0x0F];
149 }
150 return out;
151}

153Hsv toHsv(const Rgb& c)
154{
155 const double r = c.r / 255.0, g = c.g / 255.0, b = c.b / 255.0;
156 const double maxC = std::max({r, g, b});
157 const double minC = std::min({r, g, b});
158 const double delta = maxC - minC;
159
160 Hsv out;
161 out.v = maxC;
162 out.s = maxC == 0.0 ? 0.0 : delta / maxC;
163 if (delta == 0.0) return out;
164
165 double h = 0.0;
166 if (maxC == r)
167 h = std::fmod((g - b) / delta, 6.0);
168 else if (maxC == g)
169 h = (b - r) / delta + 2.0;
170 else
171 h = (r - g) / delta + 4.0;
172 h *= 60.0;
173 if (h < 0.0) h += 360.0;
174 out.h = h;
175 return out;
176}

178Rgb applyTint(const Rgb& color, double tint)
179{
180 if (tint == 0.0) return color;
181 Hls hls = toHls(color);
182 if (tint < 0.0)
183 hls.l = hls.l * (1.0 + tint);
184 else
185 hls.l = hls.l * (1.0 - tint) + tint;
186 hls.l = std::clamp(hls.l, 0.0, 1.0);
187 return fromHls(hls);
188}

199{

201
202 const Hsv hsv = toHsv(color);
